# --- OpenCV ---
# contrib가 아니라 일반 opencv-python으로 충분하다.
# ArUco는 4.7부터 contrib에서 main 저장소(objdetect)로 옮겨져 기본 패키지에 포함된다.
# (contrib를 따로 깔면 ultralytics가 요구하는 opencv-python과 같은 'cv2'를 두고 충돌한다.
#  둘 다 설치되면 나중에 깐 쪽이 이겨서 예측 불가능한 상태가 된다)
# 5.0은 아직 검증 전이라 4.x로 묶어둔다. ultralytics 요구사항(>=4.6.0)과도 호환된다.
opencv-python>=4.8.0,<5.0.0

# --- MediaPipe ---
# ★ 0.10.30 이상 필수 ★
# 이전 버전의 legacy Solutions API(mp.solutions.pose / holistic)는 패키지에서 제거되었고,
# 이 프로젝트는 새 Tasks API(PoseLandmarker)로 작성되어 있다.
# 모델 파일(.task)은 별도로 받아야 한다:  python tools/download_pose_model.py
mediapipe>=0.10.30

# --- numpy ---
# 버전을 고정하지 않는다. numpy<2.0 은 Python 3.13+ 용 wheel이 없어서
# 소스 빌드를 시도하다 컴파일러가 없으면 실패한다.
numpy>=1.24
